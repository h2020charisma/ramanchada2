119    def process(
120        self,
121        old_spe: Spectrum,
122        spe_units: Literal["cm-1", "nm", "pixel"] = "cm-1",
123        convert_back=False,
124    ):
125        new_spe = self.convert_units(old_spe, spe_units, self.model_units)
126        logger.debug(
127            "convert spe_units {} --> model units {}".format(
128                spe_units, self.model_units
129            )
130        )
131        if self.model is None:
132            return new_spe
133        elif self.enabled:
134            if isinstance(self.model, float):
135                new_spe.x = new_spe.x + self.model
136            else:
137                if isinstance(self.model, CustomPChipInterpolator):
138                    new_spe.x = self.model(new_spe.x)
139                elif isinstance(self.model, CustomPolyInterpolator):
140                    new_spe.x = self.model(new_spe.x)
141                elif isinstance(self.model, CustomRBFInterpolator):
142                    new_spe.x = self.model(new_spe.x.reshape(-1, 1))
143                    if not self.extrapolate:
144                        min_train, max_train = self.model.y.min(), self.model.y.max()
145                        # new_spe.x is the internal READ-ONLY array; np.array (unlike
146                        # np.asarray) always copies, so the masking below is legal
147                        _newx = np.array(new_spe.x, dtype=float)
148                        out_of_bounds = (_newx < min_train) | (_newx > max_train)
149                        _newx[out_of_bounds] = np.nan
150                        new_spe.x = _newx
151
152                elif isinstance(self.model, CustomCubicSplineInterpolator):
153                    new_spe.x = self.model(new_spe.x)
154
155                if np.any(np.diff(new_spe.x[np.isfinite(new_spe.x)]) <= 0):
156                    if self.nonmonotonic == "error":
157                        raise ValueError(f"Non-monotonic values detected (mode={self.nonmonotonic})")
158                    elif (self.nonmonotonic == "nan") or (self.nonmonotonic == "drop"):
159                        # this is a patch, mostly intended at extrapolation
160                        # (np.array not np.asarray: new_spe.x is read-only, we mutate below)
161                        _newx = np.array(new_spe.x, dtype=float)
162                        is_nonmonotonic = np.diff(_newx, prepend=_newx[0]) <= 0
163                        _newx[is_nonmonotonic] = np.nan
164                        new_spe.x = _newx
165                        if self.nonmonotonic == "drop":
166                            new_spe = new_spe.dropna()
167                        # we don't necessary ensure monotonicity by setting nans
168                        if np.any(np.diff(new_spe.x[np.isfinite(new_spe.x)]) <= 0):
169                            raise ValueError(f"Non-monotonic values detected (mode={self.nonmonotonic})")
170        if convert_back:
171            return self.convert_units(new_spe, self.model_units, spe_units)
172        else:
173            return new_spe

432def match_peaks(spe_pos_dict, ref_dict, spe_units, match_method="qargmin2d"):
433    _match_method = match_method
434    if spe_units == "pixel" and match_method != "qargmin2d":
435        _match_method = "dynamicp"
436    logger.debug(f"{_match_method} spe_pos_dict {spe_pos_dict}, \nref {ref_dict}")
437    if _match_method == "cluster":
438        x_spe, x_reference, x_distance, _ = match_peaks_cluster(
439            spe_pos_dict, ref_dict,
440            # _filter_range = self.spe_units != "pixel"
441        )
442        x_inliers, y_inliers, inlier_mask = qmatch.robust_poly_residual_filter(
443            x_spe, x_reference, n_sigma=3
444        )
445        logger.debug(f"Outliers found {len(x_spe)-len(x_inliers)}")
446        cost_matrix = None
447        df = pd.DataFrame(
448            {
449                "spe": x_spe,
450                "reference": x_reference,
451                "distances": x_spe - x_reference,
452                "inlier_mask": inlier_mask
453            }
454        )
455        return x_inliers, y_inliers, x_inliers-y_inliers, cost_matrix, df
456    elif _match_method == "dynamicp":
457        x_spe, x_reference, cost_matrix, df = match_peaks_ready_wrapper(
458            spe_pos_dict, ref_dict,
459        )
460        x_inliers, y_inliers, inlier_mask = qmatch.robust_poly_residual_filter(
461            x_spe, x_reference, n_sigma=3
462        )
463        df["inlier_mask"] = inlier_mask
464        logger.debug(f"Outliers found {len(x_spe)-len(x_inliers)}")
465        return x_inliers, y_inliers, x_inliers - y_inliers, cost_matrix, df
466    elif _match_method == "qargmin2d":
467        x = np.array(list(spe_pos_dict.keys()))
468        y = np.array(list(ref_dict.keys()))
469        if spe_units == "pixel":
470            x_idx, y_idx = qmatch.find_closest_pairs_quantile_idx(x, y, n_sigma=3)
471        else:
472            x_idx, y_idx = find_closest_pairs_idx(x, y)
473        x_spe = x[x_idx]
474        x_reference = y[y_idx]
475        # Sort by x
476        idx = np.argsort(x_spe)
477        x_spe = x_spe[idx]
478        x_reference = x_reference[idx]
479        # iterative_linear_filter
480        x_inliers, y_inliers, inlier_mask = qmatch.robust_poly_residual_filter(
481            x_spe, x_reference, n_sigma=3
482        )
483        logger.debug(f"Outliers found {len(x_spe)-len(x_inliers)}")
484        df = pd.DataFrame(
485            {
486                "spe": x_spe,
487                "reference": x_reference,
488                "distances": x_spe - x_reference,
489                "inlier_mask": inlier_mask
490            }
491        )
492        return x_inliers, y_inliers, x_inliers-y_inliers, None, df
493    elif _match_method == "argmin2d":
494        x = np.array(list(spe_pos_dict.keys()))
495        y = np.array(list(ref_dict.keys()))
496        x_idx, y_idx = find_closest_pairs_idx(x, y)
497        x_spe = x[x_idx]
498        x_reference = y[y_idx]
499        df = pd.DataFrame(
500            {
501                "spe": x_spe,
502                "reference": x_reference,
503                "distances": x_spe - x_reference,
504                "inlier_mask": True
505            }
506        )
507        return x_spe, x_reference, x_spe - x_reference, None, df
508    elif _match_method == "assignment":  # https://en.wikipedia.org/wiki/Hungarian_algorithm
509        try:
510            x_spe, x_reference, x_distance, cost_matrix, df = match_peaks_optimized(
511                spe_pos_dict=spe_pos_dict,
512                ref=ref_dict,
513                tolerance=100, relative=False, weight_intensity=0.9
514            )
515            return x_spe, x_reference, x_distance, cost_matrix, df
516        except Exception as err:
517            logger.warning(f"{err} Reverting to monotonic match")
518            x_spe, x_reference, x_distance,  df = match_peaks_monotonic(
519                spe_pos_dict=spe_pos_dict,
520                ref=ref_dict,
521                tolerance=None, relative=False, weight_intensity=0.25
522            )
523            return x_spe, x_reference, x_distance, None, df
524    else:  # self.match_method == "monotonic":
