ramanchada2.protocols.calibration.resolution

CWA 18133:2024 Figure 2, sections 3 & 4 — resolution curves.

Section 3: the spectral distribution curve (Raman-shift width per pixel of the calibrated x-axis) and the pixel resolution curve (Gaussian-fit neon peak FWHM vs position on the calibrated Raman-shift axis). Section 4: the spectral resolution from the calcite ~1085.91 cm-1 peak FWHM (Voigt fit, ASTM E2529 formula), the spectral-resolution curve and the SpeD:SRes curve.

This is the per-spectrum core of the CHARISMA/VAMAS spectraframe_resolution task, lifted out of its batch harness so it can be called on a single calibrated neon spectrum (plus an optional calcite spectrum) given a fitted ~ramanchada2.protocols.calibration.calibration_model.CalibrationModel. It takes a live model, not a file, so it is indifferent to how the model was persisted (pickle or JSON to_dict/from_dict).

Only the x-calibration model is used; y-calibration is intentionally not applied (CWA sections 3-4 are x-axis only).

 40# CWA 18133 Table 7 (RR1 calcite study); ASTM E2529 uses the ~1085 cm-1 band
 41CALCITE_REF_CM1 = 1085.91
 42CALCITE_WINDOW_CM1 = 100
 43# CWA 18133 Table 1 boundary of use
 44PIXEL_RESOLUTION_BOUNDARY_NM = 0.8
 45# max distance between a fitted peak and a NIST neon line to accept it
 46NEON_MATCH_TOL_CM1 = 10
 47# min fitted neon peaks required to trust a resolution curve (below this the
 48# polynomial is under-determined and the curve is not drawn / not enveloped)
 49MIN_NEON_PEAKS = 6
 50# resolution curves are only evaluated within the neon peak span, widened by
 51# this fraction of the span (small margin, no far extrapolation)
 52CURVE_MARGIN_FRAC = 0.05
 53# raw x-axis spacing with relative spread below this is a vendor-resampled
 54# (uniform) grid: the spectral distribution then shows the export grid, not
 55# the physical detector pixels
 56UNIFORM_GRID_REL_TOL = 0.01
 57# the spectral resolution cannot be meaningfully better than the neon-derived
 58# pixel resolution (neon lines have ~zero intrinsic width, so their FWHM *is*
 59# the instrument function); a laser-effect ratio below this bound - allowing
 60# for the ~20% stated accuracy of the ASTM E2529 formula - means the calcite
 61# fit is defective and the rescale is not applied
 62SRES_MIN_RATIO = 0.8
 63# ASTM E2529 spectral-resolution formula as applied in the VAMAS P6 analysis
 64# (Erlon 2026-05-04, cross-checked against Ankit and ASTM): the calcite 1085
 65# band's intrinsic width is subtracted (offset 0.684) and the instrument
 66# response scaled (slope 1.0209), so SRes = (FWHM - 0.684) / 1.0209. FWHM 8.011
 67# -> 7.177 cm-1. (Not FWHM/0.684 - 1.029, which swaps the two roles.)
 68E2529_OFFSET = 0.684
 69E2529_SLOPE = 1.0209
 70# default peak-finding window (VAMAS-tuned; see the calibration engine)
 71DEFAULT_FIND_KW = {"wlen": 200, "width": 1}
 72
 73_SEAM_NOTE = (
 74    "Narrow dips at a handful of regularly-spaced points in the spectral "
 75    "distribution curve mark detector segment-stitching seams already present "
 76    "in the raw spectrum, not a calibration or fitting defect - the "
 77    "pixel/spectral resolution curves are fit through the neon peaks and are "
 78    "not affected by them."

 93def detect_uniform_grid(x, rel_tol=UNIFORM_GRID_REL_TOL):
 94    """``(is_uniform, median_step)`` of a raw x-axis. A (near-)constant spacing
 95    means the vendor export was resampled onto a uniform grid."""
 96    d = np.diff(np.asarray(x, dtype=float))
 97    med = float(np.median(d))
 98    if len(d) < 2 or med == 0:
 99        return False, med
100    return bool((d.max() - d.min()) / abs(med) < rel_tol), med

103def fwhm_cm1_to_nm(center_cm1, fwhm_cm1, laser_wl):
104    """FWHM expressed on the wavelength axis at the given peak position."""
105    lo = shift_cm_1_to_abs_nm(center_cm1 - fwhm_cm1 / 2, laser_wl)
106    hi = shift_cm_1_to_abs_nm(center_cm1 + fwhm_cm1 / 2, laser_wl)
107    return abs(hi - lo)

110def spectral_distribution(spe_calibrated):
111    """CWA 18133 3.1.9: the Raman-shift width collected by pixel n, taken as
112    ``halfway(n, n+1) - halfway(n-1, n) == np.gradient``."""
113    x = spe_calibrated.x
114    return x, np.gradient(x)

117def neon_reference_cm1(laser_wl):
118    """NIST neon lines (nm) converted to Raman shift for this laser."""
119    lines_nm = np.array(sorted(rc2const.NEON_WL[laser_wl].keys()))
120    return np.sort(abs_nm_to_shift_cm_1(lines_nm, laser_wl))

123def fit_neon_peaks(spe_ne_cal, laser_wl, find_kw=None, prominence_coeff=3):
124    """Pixel-resolution points: Gaussian fit of the neon peaks on the calibrated
125    Raman-shift axis. Only candidate groups near a NIST neon line are fitted, and
126    each reference line keeps its single best (highest) fitted peak - otherwise
127    noise bumps distort the resolution curve."""
128    ref_cm1 = neon_reference_cm1(laser_wl)
129    find_kw = dict(find_kw or DEFAULT_FIND_KW)
130    find_kw["prominence"] = spe_ne_cal.y_noise_MAD() * prominence_coeff
131    cand = spe_ne_cal.find_peak_multipeak(**find_kw)
132    groups = [
133        g
134        for g in cand
135        if np.min(np.abs(np.asarray(g.positions)[:, None] - ref_cm1[None, :]))
136        < NEON_MATCH_TOL_CM1
137    ]
138    if not groups:
139        return pd.DataFrame(columns=["center", "fwhm", "fwhm_stderr", "height"])
140    fitres = spe_ne_cal.fit_peak_multimodel(
141        profile="Gaussian",
142        candidates=ListPeakCandidateMultiModel(root=groups),
143        no_fit=False,
144        bound_centers_to_group=True,
145        vary_baseline=False,
146    )
147    df_peaks = fitres.to_dataframe_peaks()
148    df_peaks = df_peaks.loc[
149        np.isfinite(df_peaks["fwhm"])
150        & (df_peaks["fwhm"] > 0)
151        & np.isfinite(df_peaks["center"])
152        & (df_peaks["center"] >= min(spe_ne_cal.x))
153        & (df_peaks["center"] <= max(spe_ne_cal.x))
154    ]
155    if "fwhm_stderr" in df_peaks.columns:
156        df_peaks = df_peaks.loc[
157            df_peaks["fwhm_stderr"].isna()
158            | (df_peaks["fwhm_stderr"] < df_peaks["fwhm"])
159        ]
160    if df_peaks.empty:
161        return df_peaks
162    # one fitted peak per NIST line: nearest line within tolerance, best height wins
163    idx = np.argmin(
164        np.abs(df_peaks["center"].values[:, None] - ref_cm1[None, :]), axis=1
165    )
166    df_peaks = df_peaks.assign(ref_line=ref_cm1[idx])
167    df_peaks = df_peaks.loc[
168        (df_peaks["center"] - df_peaks["ref_line"]).abs() < NEON_MATCH_TOL_CM1
169    ]
170    df_peaks = (
171        df_peaks.sort_values("height", ascending=False)
172        .groupby("ref_line", as_index=False)
173        .first()
174    )
175    return df_peaks.sort_values(by="center")

178def fit_pixel_resolution_curve(centers, fwhms, degree=2):
179    """CWA 18133 3.1.5: a function fit of FWHM vs neon peak position.
180
181    Requires at least ``MIN_NEON_PEAKS`` points, and does one round of MAD-based
182    outlier rejection so a single mis-fit neon peak does not distort the curve.
183    Returns ``(poly, fit_lo, fit_hi)`` or ``(None, None, None)``."""
184    centers = np.asarray(centers, dtype=float)
185    fwhms = np.asarray(fwhms, dtype=float)
186    if len(centers) < MIN_NEON_PEAKS:
187        return None, None, None
188    deg = min(degree, len(centers) - 1)
189    if deg < 1:
190        return None, None, None
191    poly = np.poly1d(np.polyfit(centers, fwhms, deg))
192    resid = fwhms - poly(centers)
193    mad = np.median(np.abs(resid - np.median(resid)))
194    if mad > 0:
195        keep = np.abs(resid - np.median(resid)) <= 3 * 1.4826 * mad
196        if keep.sum() >= MIN_NEON_PEAKS and keep.sum() < len(centers):
197            centers, fwhms = centers[keep], fwhms[keep]
198            deg = min(degree, len(centers) - 1)
199            poly = np.poly1d(np.polyfit(centers, fwhms, deg))
200    return poly, float(centers.min()), float(centers.max())

203def clip_to_range(x, lo, hi):
204    """Boolean mask of ``x`` within ``[lo, hi]`` widened by ``CURVE_MARGIN_FRAC``."""
205    margin = (hi - lo) * CURVE_MARGIN_FRAC
206    return (x >= lo - margin) & (x <= hi + margin)

209def fit_calcite_1085(spe_cal_calibrated, find_kw=None, prominence_coeff=3):
210    """Voigt fit of the calcite ~1085.91 cm-1 peak (CWA Figure 2, Section 4).
211    Falls back to Gaussian when the Voigt fit aborts (lmfit occasionally
212    generates NaN model values); E2529 accepts mixed Gaussian/Lorentzian.
213    Returns ``(peak_series_or_None, prepared_spectrum)``."""
214    find_kw = dict(find_kw or DEFAULT_FIND_KW)
215    spe = spe_cal_calibrated.dropna().trim_axes(
216        method="x-axis",
217        boundaries=(
218            CALCITE_REF_CM1 - CALCITE_WINDOW_CM1,
219            CALCITE_REF_CM1 + CALCITE_WINDOW_CM1,
220        ),
221    )
222    spe.y = spe.y - np.min(spe.y)
223    spe = spe.subtract_baseline_rc1_snip(niter=40)
224    fitres = None
225    profile = None
226    for profile in ("Voigt", "Gaussian"):
227        try:
228            _find_kw = dict(find_kw)
229            _find_kw["prominence"] = spe.y_noise_MAD() * prominence_coeff
230            cand = spe.find_peak_multipeak(**_find_kw)
231            fitres = spe.fit_peak_multimodel(
232                profile=profile,
233                candidates=cand,
234                no_fit=False,
235                bound_centers_to_group=True,
236                vary_baseline=False,
237            )
238            break
239        except Exception:  # noqa: BLE001 - try the next profile, then give up
240            fitres = None
241    if fitres is None:
242        return None, spe
243    df_peaks = fitres.to_dataframe_peaks()
244    df_peaks = df_peaks.assign(profile=profile)
245    df_peaks = df_peaks.loc[np.isfinite(df_peaks["fwhm"]) & (df_peaks["fwhm"] > 0)]
246    if df_peaks.empty:
247        return None, spe
248    df_peaks = df_peaks.iloc[(df_peaks["center"] - CALCITE_REF_CM1).abs().argsort()]
249    peak = df_peaks.iloc[0]
250    if abs(peak["center"] - CALCITE_REF_CM1) > 20:
251        return None, spe
252    return peak, spe

255def spectral_resolution_e2529(fwhm_1085, offset=E2529_OFFSET, slope=E2529_SLOPE):
256    """ASTM E2529 spectral resolution: ``(FWHM_1085 - 0.684) / 1.0209``."""
257    return (fwhm_1085 - offset) / slope

260def _eval_clipped(curve, x_sped, in_range):
261    """Evaluate a curve only within the neon-supported range (NaN elsewhere)."""
262    if curve is None:
263        return np.full(len(x_sped), np.nan)
264    y = np.asarray(curve(x_sped), dtype=float)
265    y[~in_range] = np.nan
266    return y

382def resolution_from_calibration(
383    calmodel: "CalibrationModel",
384    spe_neon: Spectrum,
385    *,
386    neon_units: str = "cm-1",
387    spe_calcite: Spectrum | None = None,
388    calcite_units: str = "cm-1",
389    find_kw=None,
390    calcite_find_kw=None,
391    prominence_coeff: float = 3,
392    curve_fit_degree: int = 2,
393    e2529_offset: float = E2529_OFFSET,
394    e2529_slope: float = E2529_SLOPE,
395    title: str = "",
396) -> ResolutionResult:
397    """CWA 18133 sections 3 & 4 for one x-calibration and its neon (+ calcite).
398
399    ``calmodel`` is a fitted :class:`CalibrationModel`; ``spe_neon`` is the raw
400    neon spectrum in ``neon_units``. The calibrated Raman-shift axis is obtained
401    with ``calmodel.apply_calibration_x`` (x-calibration only). Pass ``spe_calcite``
402    to add the ASTM E2529 spectral-resolution curve.
403    """
404    laser_wl = int(calmodel.laser_wl)
405    notes: list[str] = [_SEAM_NOTE]
406
407    # Section 3 - calibrated Raman shift axis applied to the neon spectrum
408    uniform_grid, grid_step = detect_uniform_grid(spe_neon.x)
409    if uniform_grid:
410        notes.append(_UNIFORM_GRID_NOTE)
411    spe_ne_cal = calmodel.apply_calibration_x(spe_neon, spe_units=neon_units).dropna()
412
413    x_sped, sped = spectral_distribution(spe_ne_cal)
414    ne_peaks = fit_neon_peaks(spe_ne_cal, laser_wl, find_kw, prominence_coeff)
415    pixel_res_curve, fit_lo, fit_hi = fit_pixel_resolution_curve(
416        ne_peaks["center"].values, ne_peaks["fwhm"].values, curve_fit_degree
417    )
418    in_range = (
419        clip_to_range(x_sped, fit_lo, fit_hi)
420        if pixel_res_curve is not None
421        else np.zeros(len(x_sped), bool)
422    )
423    ne_peaks = ne_peaks.assign(
424        fwhm_nm=[
425            fwhm_cm1_to_nm(c, f, laser_wl)
426            for c, f in zip(ne_peaks["center"], ne_peaks["fwhm"])
427        ]
428    )
429
430    # Section 4 - calcite spectral resolution (ASTM E2529)
431    calcite_peak, sres, ratio, sres_plausible = None, None, None, None
432    if spe_calcite is not None and pixel_res_curve is not None:
433        spe_cal_calibrated = calmodel.apply_calibration_x(
434            spe_calcite, spe_units=calcite_units
435        )
436        calcite_peak, _ = fit_calcite_1085(
437            spe_cal_calibrated, calcite_find_kw, prominence_coeff
438        )
439        if calcite_peak is not None:
440            sres = spectral_resolution_e2529(
441                calcite_peak["fwhm"], e2529_offset, e2529_slope
442            )
443            neon_fwhm_1085 = float(pixel_res_curve(calcite_peak["center"]))
444            # laser effect adjustment: scale the pixel resolution curve so it
445            # passes through the calcite spectral resolution value. A rescale
446            # that pushes the curve well below the neon-derived instrument
447            # function is physically impossible - the calcite fit is defective.
448            _ratio = sres / neon_fwhm_1085 if neon_fwhm_1085 else 0.0
449            if _ratio < SRES_MIN_RATIO or sres <= 0:
450                sres_plausible = False
451                notes.append(_IMPLAUSIBLE_NOTE)
452            else:
453                sres_plausible = True
454                ratio = _ratio
455
456    spectral_res_curve: Callable | None = (
457        None if ratio is None else (lambda x, r=ratio, c=pixel_res_curve: r * c(x))
458    )
459
460    pixel_res = _eval_clipped(pixel_res_curve, x_sped, in_range)
461    spectral_res = _eval_clipped(spectral_res_curve, x_sped, in_range)
462    with np.errstate(divide="ignore", invalid="ignore"):
463        sped_sres = sped / spectral_res
464    sped_sres = np.where(np.isfinite(spectral_res), sped_sres, np.nan)
465
466    curve_ok = pixel_res_curve is not None
467    curve_monotonic = None
468    if curve_ok:
469        xr = np.linspace(fit_lo, fit_hi, 50)
470        curve_monotonic = bool(np.all(np.diff(pixel_res_curve(xr)) >= -1e-9))
471    max_fwhm_nm = ne_peaks["fwhm_nm"].max() if not ne_peaks.empty else np.nan
472
473    return ResolutionResult(
474        laser_wl=laser_wl,
475        raman_shift=np.asarray(x_sped),
476        sped=np.asarray(sped),
477        pixel_res=pixel_res,
478        spectral_res=spectral_res,
479        sped_sres=sped_sres,
480        neon_peaks=ne_peaks,
481        n_neon_peaks=len(ne_peaks),
482        curve_ok=curve_ok,
483        curve_monotonic=curve_monotonic,
484        fit_lo=fit_lo,
485        fit_hi=fit_hi,
486        neon_fwhm_median=(
487            float(ne_peaks["fwhm"].median()) if not ne_peaks.empty else None
488        ),
489        max_neon_fwhm_nm=float(max_fwhm_nm) if np.isfinite(max_fwhm_nm) else None,

def fit_neon_peaks(spe_ne_cal, laser_wl, find_kw=None, prominence_coeff=3): 
124def fit_neon_peaks(spe_ne_cal, laser_wl, find_kw=None, prominence_coeff=3):
125    """Pixel-resolution points: Gaussian fit of the neon peaks on the calibrated
126    Raman-shift axis. Only candidate groups near a NIST neon line are fitted, and
127    each reference line keeps its single best (highest) fitted peak - otherwise
128    noise bumps distort the resolution curve."""
129    ref_cm1 = neon_reference_cm1(laser_wl)
130    find_kw = dict(find_kw or DEFAULT_FIND_KW)
131    find_kw["prominence"] = spe_ne_cal.y_noise_MAD() * prominence_coeff
132    cand = spe_ne_cal.find_peak_multipeak(**find_kw)
133    groups = [
134        g
135        for g in cand
136        if np.min(np.abs(np.asarray(g.positions)[:, None] - ref_cm1[None, :]))
137        < NEON_MATCH_TOL_CM1
138    ]
139    if not groups:
140        return pd.DataFrame(columns=["center", "fwhm", "fwhm_stderr", "height"])
141    fitres = spe_ne_cal.fit_peak_multimodel(
142        profile="Gaussian",
143        candidates=ListPeakCandidateMultiModel(root=groups),
144        no_fit=False,
145        bound_centers_to_group=True,
146        vary_baseline=False,
147    )
148    df_peaks = fitres.to_dataframe_peaks()
149    df_peaks = df_peaks.loc[
150        np.isfinite(df_peaks["fwhm"])
151        & (df_peaks["fwhm"] > 0)
152        & np.isfinite(df_peaks["center"])
153        & (df_peaks["center"] >= min(spe_ne_cal.x))
154        & (df_peaks["center"] <= max(spe_ne_cal.x))
155    ]
156    if "fwhm_stderr" in df_peaks.columns:
157        df_peaks = df_peaks.loc[
158            df_peaks["fwhm_stderr"].isna()
159            | (df_peaks["fwhm_stderr"] < df_peaks["fwhm"])
160        ]
161    if df_peaks.empty:
162        return df_peaks
163    # one fitted peak per NIST line: nearest line within tolerance, best height wins
164    idx = np.argmin(
165        np.abs(df_peaks["center"].values[:, None] - ref_cm1[None, :]), axis=1
166    )
167    df_peaks = df_peaks.assign(ref_line=ref_cm1[idx])
168    df_peaks = df_peaks.loc[
169        (df_peaks["center"] - df_peaks["ref_line"]).abs() < NEON_MATCH_TOL_CM1
170    ]
171    df_peaks = (
172        df_peaks.sort_values("height", ascending=False)
173        .groupby("ref_line", as_index=False)
174        .first()
175    )
176    return df_peaks.sort_values(by="center")

def fit_pixel_resolution_curve(centers, fwhms, degree=2): 
179def fit_pixel_resolution_curve(centers, fwhms, degree=2):
180    """CWA 18133 3.1.5: a function fit of FWHM vs neon peak position.
181
182    Requires at least ``MIN_NEON_PEAKS`` points, and does one round of MAD-based
183    outlier rejection so a single mis-fit neon peak does not distort the curve.
184    Returns ``(poly, fit_lo, fit_hi)`` or ``(None, None, None)``."""
185    centers = np.asarray(centers, dtype=float)
186    fwhms = np.asarray(fwhms, dtype=float)
187    if len(centers) < MIN_NEON_PEAKS:
188        return None, None, None
189    deg = min(degree, len(centers) - 1)
190    if deg < 1:
191        return None, None, None
192    poly = np.poly1d(np.polyfit(centers, fwhms, deg))
193    resid = fwhms - poly(centers)
194    mad = np.median(np.abs(resid - np.median(resid)))
195    if mad > 0:
196        keep = np.abs(resid - np.median(resid)) <= 3 * 1.4826 * mad
197        if keep.sum() >= MIN_NEON_PEAKS and keep.sum() < len(centers):
198            centers, fwhms = centers[keep], fwhms[keep]
199            deg = min(degree, len(centers) - 1)
200            poly = np.poly1d(np.polyfit(centers, fwhms, deg))
201    return poly, float(centers.min()), float(centers.max())
