197        >>> fig, ax = plt.subplots(1, 1, figsize=(15,4))
198        >>> spe_srm.plot(ax=ax)
199        >>> spe_to_correct.plot(ax=ax)
200        >>> spe_ycalibrated = ycal.process(spe_to_correct)
201        >>> spe_ycalibrated.plot(label="y-calibrated",color="green",ax=ax.twinx())

205        self,
206        laser_wl,
207        reference_spe_xcalibrated,
208        certificate: YCalibrationCertificate,
209        model_method: str = "certificate",
210        fit_order=None,
211        normalize: bool = True,

213        super(YCalibrationComponent, self).__init__(

233    def _trimmed_reference(self):
234        # certificate.raman_shift is optional; without a certified range there is
235        # nothing to trim to, so use the full measured reference spectrum instead
236        # of letting trim_axes(boundaries=None) raise
237        if self.ref.raman_shift is None:
238            return self.spe
239        return self.spe.trim_axes(method="x-axis", boundaries=self.ref.raman_shift)

241    def _build_model(self):
242        tmp = self._trimmed_reference()
243        if self.model_method == "certificate":
244            fitted = self._fit_reference(tmp)
245            if fitted is not None:
246                return fitted
247        return CustomPChipInterpolator(tmp.x, tmp.y)

249    def _fit_reference(self, tmp):
250        """Fit the measured reference with the certificate's own functional form.
251
252        Polynomial certificates -> a polynomial of the same order (linear least
253        squares); other forms (the SRM log-Gaussian) -> a nonlinear fit of the
254        certificate equation, seeded by the certificate's own parameters. The
255        analytic fit denoises the reference without smoothing. Returns a
256        ``ParametricModel``, or ``None`` to fall back to the raw PCHIP.
257        """
258        from .interpolators import ParametricModel
259        from scipy.optimize import curve_fit
260
261        x = np.asarray(tmp.x, dtype=float)
262        y = np.asarray(tmp.y, dtype=float)
263        finite = np.isfinite(x) & np.isfinite(y)
264        x, y = x[finite], y[finite]
265        if len(x) < 6:
266            logger.warning("Y calibration: too few reference points to fit; using PCHIP")
267            return None
268        if self.normalize:
269            ymax = float(np.max(np.abs(y)))
270            if ymax > 0:
271                y = y / ymax
272
273        # fit_order forces a polynomial of that order; otherwise the certificate's
274        # own order (polynomial) or None (fit the certificate's equation directly)
275        order = self.fit_order if self.fit_order is not None else self.ref.polynomial_order
276        try:
277            if order is not None:
278                order = int(order)
279                coef_desc = np.polyfit(x, y, order)
280                names = [f"A{i}" for i in range(order + 1)]
281                terms = ["A0"] + [f"A{i}*x**{i}" for i in range(1, order + 1)]
282                model = ParametricModel(
283                    " + ".join(terms), names, list(coef_desc[::-1])
284                )
285            else:
286                names = self.ref.param_names
287                popt, _ = curve_fit(
288                    self.ref.model_function(), x, y, p0=self.ref.param_values,
289                    maxfev=20000,
290                )
291                model = ParametricModel(self.ref.equation, names, list(popt))
292            probe = np.asarray(model(x), dtype=float)
293            if not np.all(np.isfinite(probe)):
294                logger.warning("Y calibration: fitted reference not finite; using PCHIP")
295                return None
296            return model
297        except Exception as err:  # noqa: BLE001 - degrade to PCHIP, never crash
298            logger.warning(f"Y calibration: reference fit failed ({err}); using PCHIP")
299            return None

301    def derive_model(self, find_kw=None, fit_peaks_kw=None, should_fit=True, name=None):
302        self.model = self._build_model()

304    def safe_divide(self, spe_to_correct, spe_reference_resampled):
305        numerator = spe_to_correct.y
306        # numerator_noise = spe_to_correct.y_noise
307
308        scaling_denominator = spe_reference_resampled.y / self.ref.Y(
309            spe_reference_resampled.x
310        )
311        # print(np.median(scaling_denominator), np.mean(scaling_denominator), np.std(scaling_denominator))
312
313        # denominator_noise = spe_reference_resampled.y_noise
314        denominator = spe_reference_resampled.y
315        # Create a mask for dividing only where value is above noise !
316        # mask = (abs(scaling_denominator) > 0) & (kind_of_snr > 0.9)
317        # mask =  (abs(denominator) > abs(denominator_noise)) &
318        mask = (abs(scaling_denominator) > 0) & (numerator > 0) & (denominator > 0)
319        # & (abs(numerator) > numerator_noise) & (abs(scaling_denominator) > 0)
320        # & (abs(denominator-numerator) > min(denominator_noise,numerator_noise))
321        result = np.zeros_like(numerator)
322        # Perform division where mask is true
323        result[mask] = numerator[mask] / scaling_denominator[mask]
324        return result
325
326    def safe_mask(self, spe_to_correct, spe_reference_resampled):
327        ref_noise = spe_reference_resampled.y_noise_MAD()
328        mask = (spe_reference_resampled.y >= 0) & (
329            abs(spe_reference_resampled.y) > ref_noise
330        )
331        # Outside the certified range, self.model (CustomPChipInterpolator) falls
332        # back to a unit-slope extrapolation meant for x-calibration curves, not
333        # measured reference intensity -- exclude those points rather than treat
334        # the extrapolated value as a real intensity factor.
335        if self.ref.raman_shift is not None:
336            lo, hi = self.ref.raman_shift
337            x = spe_reference_resampled.x
338            mask = mask & (x >= lo) & (x <= hi)
339        return mask
340
341    def safe_factor(self, spe_to_correct, spe_reference_resampled):
342        numerator = spe_to_correct.y
343        # numerator_noise = spe_to_correct.y_noise
344
345        Y = self.ref.Y(spe_reference_resampled.x)
346        mask = self.safe_mask(spe_to_correct, spe_reference_resampled)
347        if mask is None:
348            scaling_factor = Y / spe_reference_resampled.y
349        else:
350            scaling_factor = np.zeros_like(spe_reference_resampled.y)
351            scaling_factor[mask] = Y[mask] / spe_reference_resampled.y[mask]
352
353        result = numerator * scaling_factor
354        return result

356    def process(self, old_spe: Spectrum, spe_units="nm", convert_back=False):
357        # resample using probability density function
358        _tmp = self.model(old_spe.x)
359        spe_reference_resampled = Spectrum(old_spe.x, _tmp)
360        # new_spe = Spectrum(old_spe.x,self.safe_divide(old_spe,spe_reference_resampled))
361        new_spe = Spectrum(
362            old_spe.x, self.safe_factor(old_spe, spe_reference_resampled)
363        )
364        return new_spe

242    def _build_model(self):
243        tmp = self._trimmed_reference()
244        if self.model_method == "certificate":

250    def _fit_reference(self, tmp):
251        """Fit the measured reference with the certificate's own functional form.
252
253        Polynomial certificates -> a polynomial of the same order (linear least
254        squares); other forms (the SRM log-Gaussian) -> a nonlinear fit of the
255        certificate equation, seeded by the certificate's own parameters. The
256        analytic fit denoises the reference without smoothing. Returns a
257        ``ParametricModel``, or ``None`` to fall back to the raw PCHIP.
258        """
259        from .interpolators import ParametricModel
260        from scipy.optimize import curve_fit
261
262        x = np.asarray(tmp.x, dtype=float)
263        y = np.asarray(tmp.y, dtype=float)
264        finite = np.isfinite(x) & np.isfinite(y)
265        x, y = x[finite], y[finite]
266        if len(x) < 6:
267            logger.warning("Y calibration: too few reference points to fit; using PCHIP")
268            return None
269        if self.normalize:
270            ymax = float(np.max(np.abs(y)))
271            if ymax > 0:
272                y = y / ymax
273
274        # fit_order forces a polynomial of that order; otherwise the certificate's
275        # own order (polynomial) or None (fit the certificate's equation directly)
276        order = self.fit_order if self.fit_order is not None else self.ref.polynomial_order
277        try:
278            if order is not None:
279                order = int(order)
280                coef_desc = np.polyfit(x, y, order)
281                names = [f"A{i}" for i in range(order + 1)]
282                terms = ["A0"] + [f"A{i}*x**{i}" for i in range(1, order + 1)]
283                model = ParametricModel(
284                    " + ".join(terms), names, list(coef_desc[::-1])
285                )
286            else:
287                names = self.ref.param_names
288                popt, _ = curve_fit(
289                    self.ref.model_function(), x, y, p0=self.ref.param_values,
290                    maxfev=20000,
291                )
292                model = ParametricModel(self.ref.equation, names, list(popt))
293            probe = np.asarray(model(x), dtype=float)
294            if not np.all(np.isfinite(probe)):
295                logger.warning("Y calibration: fitted reference not finite; using PCHIP")
296                return None
297            return model
298        except Exception as err:  # noqa: BLE001 - degrade to PCHIP, never crash
299            logger.warning(f"Y calibration: reference fit failed ({err}); using PCHIP")
300            return None

305    def safe_divide(self, spe_to_correct, spe_reference_resampled):
306        numerator = spe_to_correct.y
307        # numerator_noise = spe_to_correct.y_noise
308
309        scaling_denominator = spe_reference_resampled.y / self.ref.Y(
310            spe_reference_resampled.x
311        )
312        # print(np.median(scaling_denominator), np.mean(scaling_denominator), np.std(scaling_denominator))
313
314        # denominator_noise = spe_reference_resampled.y_noise
315        denominator = spe_reference_resampled.y
316        # Create a mask for dividing only where value is above noise !
317        # mask = (abs(scaling_denominator) > 0) & (kind_of_snr > 0.9)
318        # mask =  (abs(denominator) > abs(denominator_noise)) &
319        mask = (abs(scaling_denominator) > 0) & (numerator > 0) & (denominator > 0)
320        # & (abs(numerator) > numerator_noise) & (abs(scaling_denominator) > 0)
321        # & (abs(denominator-numerator) > min(denominator_noise,numerator_noise))
322        result = np.zeros_like(numerator)
323        # Perform division where mask is true
324        result[mask] = numerator[mask] / scaling_denominator[mask]
325        return result

327    def safe_mask(self, spe_to_correct, spe_reference_resampled):
328        ref_noise = spe_reference_resampled.y_noise_MAD()
329        mask = (spe_reference_resampled.y >= 0) & (
330            abs(spe_reference_resampled.y) > ref_noise
331        )
332        # Outside the certified range, self.model (CustomPChipInterpolator) falls
333        # back to a unit-slope extrapolation meant for x-calibration curves, not
334        # measured reference intensity -- exclude those points rather than treat
335        # the extrapolated value as a real intensity factor.
336        if self.ref.raman_shift is not None:
337            lo, hi = self.ref.raman_shift
338            x = spe_reference_resampled.x
339            mask = mask & (x >= lo) & (x <= hi)
340        return mask

342    def safe_factor(self, spe_to_correct, spe_reference_resampled):
343        numerator = spe_to_correct.y
344        # numerator_noise = spe_to_correct.y_noise
345
346        Y = self.ref.Y(spe_reference_resampled.x)
347        mask = self.safe_mask(spe_to_correct, spe_reference_resampled)

352            scaling_factor[mask] = Y[mask] / spe_reference_resampled.y[mask]

355        return result
