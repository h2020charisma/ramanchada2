 35def quantile_map(x, y, q=None):
 36    """
 37    Build coarse linear map using quantiles.
 38    No normalization needed!
 39    """
 40    if q is None:
 41        q = [0.0, 0.25, 0.5, 0.75, 1.0]
 42
 43    qx = np.quantile(x, q)
 44    qy = np.quantile(y, q)
 45    a = np.polyfit(qx, qy, deg=1)
 46
 47    poly = np.poly1d(a)
 48    logger.info(f"  Quantile map: y = {a[0]:.6f}*x + {a[1]:.6f}")
 49
 50    return poly

 53def estimate_median_limit_from_data(A, n_sigma=3.0):
 54    """
 55    Estimate median_limit from distance distribution.
 56    Works directly with physical units (nm).
 57    """
 58    min_dist_per_ref = np.min(A, axis=1)
 59    min_dist_per_peak = np.min(A, axis=0)
 60    all_min_dist = np.concatenate([min_dist_per_ref, min_dist_per_peak])
 61
 62    med = np.median(all_min_dist)
 63    mad = np.median(np.abs(all_min_dist - med))
 64
 65    logger.info(f"\nAdaptive median_limit: Median distance: {med:.2f} nm MAD: {mad:.2f} nm")
 66
 67    if med < 1e-10 or mad < 1e-10:
 68        return 10.0
 69
 70    median_limit = 1.0 + n_sigma * (mad / med)
 71    print(f"  median_limit = {median_limit:.2f}")
 72
 73    return median_limit

 76def linear_residual_filter(x, y, n_sigma=3.0):
 77    """
 78    Filter outliers using linear fit residuals.
 79    Works directly in physical units.
 80    """
 81    if len(x) < 2:
 82        return x, y, np.ones(len(x), dtype=bool)
 83
 84    a, b = np.polyfit(x, y, deg=1)
 85    y_pred = a * x + b
 86    resid = y - y_pred
 87
 88    med_resid = np.median(resid)
 89    mad = np.median(np.abs(resid - med_resid))
 90
 91    if mad < 1e-10:
 92        threshold = np.percentile(np.abs(resid), 95)
 93        print(f"  Linear filter: MAD too small, using 95th percentile threshold={threshold:.2f} nm")
 94    else:
 95        threshold = n_sigma * mad
 96        print(f"  Linear filter: MAD={mad:.2f} nm, threshold={threshold:.2f} nm")
 97
 98    mask = np.abs(resid - med_resid) < threshold
 99    print(f"  Kept {mask.sum()}/{len(x)} inliers")
100
101    return x[mask], y[mask], mask

104def robust_poly_residual_filter(x, y, deg=2, n_sigma=3.0, n_iter=300, seed=0):
105    """Curve-aware robust outlier filter for matched (measured, reference) pairs.
106
107    ``linear_residual_filter`` fits a *straight line* and rejects points by their
108    residual from it. But the measured->reference relation of a Raman spectrograph is a
109    gently curved dispersion, so correctly-matched lines near the laser deviate from the
110    global straight line and get discarded. Dropping them shrinks the calibration anchor
111    span, which in turn pushes the Silicon laser-zeroing peak (and low-wavenumber sample
112    peaks) *outside* the anchor range where they are extrapolated -> the whole cm-1 axis
113    is mis-scaled and calibration worsens the spectra.
114
115    This filter keeps the largest RANSAC consensus of points lying within an adaptive band
116    of a degree-``deg`` fit, so on-trend (curved) near-laser lines survive while genuine
117    mismatches are still rejected. Returns ``(x_inliers, y_inliers, mask)`` with ``mask``
118    aligned to the input order, matching ``linear_residual_filter``.
119    """
120    x = np.asarray(x, dtype=float)
121    y = np.asarray(y, dtype=float)
122    n = len(x)
123    if n <= deg + 2:
124        return x, y, np.ones(n, dtype=bool)
125
126    xc = x - x.mean()  # centre for numerical stability of polyfit
127    # Initial (generous) acceptance band for the RANSAC search: n_sigma * robust-sigma of
128    # residuals to a curve-aware (deg) fit, floored by a fraction of the reference spacing
129    # so a near-linear region with a tiny MAD does not clip the genuinely curved tails.
130    c_all = np.polyfit(xc, y, deg)
131    r_all = y - np.polyval(c_all, xc)
132    mad = np.median(np.abs(r_all - np.median(r_all)))
133    spacing = np.median(np.abs(np.diff(np.sort(y))))
134    thr = max(n_sigma * 1.4826 * mad, 0.25 * spacing)
135    if thr <= 0:
136        thr = np.percentile(np.abs(r_all), 95)
137
138    rng = np.random.default_rng(seed)
139    best = np.zeros(n, dtype=bool)
140    for _ in range(n_iter):
141        idx = rng.choice(n, deg + 1, replace=False)
142        try:
143            c = np.polyfit(xc[idx], y[idx], deg)
144        except Exception:
145            continue
146        m = np.abs(y - np.polyval(c, xc)) < thr
147        if m.sum() > best.sum():
148            best = m
149    if best.sum() >= deg + 2:  # refit on the consensus and re-select
150        c = np.polyfit(xc[best], y[best], deg)
151        best = np.abs(y - np.polyval(c, xc)) < thr
152
153    # Doublet-misassignment split: when the matcher alternates between the components of
154    # blended reference doublets, the consensus residuals form TWO parallel populations
155    # (~a fraction of a line spacing apart). Fitting through both tilts the model; keep
156    # the majority population, which the sample-peak residuals confirm as the correct
157    # assignment (the minority would shift the axis by tens of cm-1).
158    if best.sum() >= 2 * (deg + 2):
159        c = np.polyfit(xc[best], y[best], deg)
160        rb = np.sort((y - np.polyval(c, xc))[best])
161        split_gap, split_at = 0.0, None
162        for k in range(deg + 2, len(rb) - (deg + 2) + 1):
163            a, b = rb[:k], rb[k:]
164            gap = b[0] - a[-1]
165            scatter = max(np.median(np.abs(a - np.median(a))),
166                          np.median(np.abs(b - np.median(b))), 1e-6)
167            if gap > 3.0 * 1.4826 * scatter and gap > split_gap:
168                split_gap, split_at = gap, (a[-1] + b[0]) / 2.0
169        if split_at is not None:
170            r_all_c = y - np.polyval(c, xc)
171            lower = best & (r_all_c < split_at)
172            upper = best & (r_all_c >= split_at)
173            major = lower if lower.sum() >= upper.sum() else upper
174            if major.sum() >= deg + 2:
175                print(f"  Robust deg-{deg} filter: bimodal residuals (gap={split_gap:.3f}), "
176                      f"keeping majority {major.sum()}/{best.sum()}")
177                best = major
178
179    # Tighten the band to the consensus scatter. The RANSAC band is floored by the
180    # reference line spacing, so near-miss mismatches (a fraction of a line spacing off,
181    # e.g. spurious peaks matched to the nearest available line) survive the search and
182    # tilt the subsequent model fit -- the exact "slope error amplified by Si zeroing"
183    # failure. The consensus residual MAD is the real measurement noise; rescale to it,
184    # with a small floor so curved tails beyond the deg-fit are not clipped.
185    if best.sum() >= deg + 2:
186        c = np.polyfit(xc[best], y[best], deg)
187        r = y - np.polyval(c, xc)
188        mad_c = np.median(np.abs(r[best] - np.median(r[best])))
189        thr2 = min(thr, max(n_sigma * 1.4826 * mad_c, 0.05 * spacing))
190        tight = np.abs(r) < thr2
191        if tight.sum() >= deg + 2:
192            c = np.polyfit(xc[tight], y[tight], deg)  # final refit on the tightened set
193            tight = np.abs(y - np.polyval(c, xc)) < thr2
194            if tight.sum() >= deg + 2:
195                best, thr = tight, thr2
196
197    print(f"  Robust deg-{deg} filter: band={thr:.3f}, kept {best.sum()}/{n} inliers")
198    return x[best], y[best], best

201def iterative_linear_filter(x, y, n_sigma=3.0, max_iter=5):
202    """
203    Iteratively fit linear model and remove outliers.
204
205    Args:
206        x, y: Matched pairs (normalized)
207        n_sigma: Number of MAD sigmas
208        max_iter: Maximum iterations
209
210    Returns:
211        x_inliers, y_inliers, mask
212    """
213    mask = np.ones(len(x), dtype=bool)
214
215    for iteration in range(max_iter):
216        x_fit = x[mask]
217        y_fit = y[mask]
218
219        if len(x_fit) < 2:
220            print(f"  Iteration {iteration}: Too few points, stopping")
221            break
222
223        # Fit linear model on current inliers
224        a, b = np.polyfit(x_fit, y_fit, deg=1)
225
226        # Compute residuals on ALL points
227        y_pred = a * x + b
228        resid = y - y_pred
229
230        # MAD on current inliers
231        resid_inliers = resid[mask]
232        med = np.median(resid_inliers)
233        mad = np.median(np.abs(resid_inliers - med))
234
235        if mad < 1e-10:
236            print(f"  Iteration {iteration}: Converged (MAD={mad:.2e})")
237            break
238
239        threshold = n_sigma * mad
240        new_mask = np.abs(resid - med) < threshold
241
242        n_removed = (mask & ~new_mask).sum()
243        print(f"  Iteration {iteration}: {new_mask.sum()}/{len(x)} inliers, "
244              f"removed {n_removed}, MAD={mad:.6f}")
245
246        if np.array_equal(mask, new_mask):
247            print("  Converged (mask unchanged)")
248            break
249
250        mask = new_mask
251
252    return x[mask], y[mask], mask

255def universal_dispersion_calibration(
256    peaks_measured,
257    reference_lines,
258    median_limit=None,
259    n_sigma_match=3.0,
260    outlier_method='linear',
261    n_sigma_outlier=3.0,
262    use_quantile_map=True,
263    interpolator="poly"
264):
265    """
266    Universal dispersion calibration WITHOUT normalization.
267    """
268
269    # --- STEP 1: MATCHING ---
270    print("\n=== STEP 1: MATCHING ===")
271
272    if use_quantile_map:
273        print("Using quantile map for coarse alignment...")
274        f0 = quantile_map(peaks_measured, reference_lines)  # No normalization!
275        A = np.abs(reference_lines[:, None] - f0(peaks_measured)[None, :])
276    else:
277        print("Direct matching (assuming same units)...")
278        A = np.abs(reference_lines[:, None] - peaks_measured[None, :])
279
280    # Distance statistics
281    min_distances = np.min(A, axis=1)
282    print(f"Distance range: [{min_distances.min():.2f}, {min_distances.max():.2f}] nm")
283
284    if median_limit is None:
285        median_limit = estimate_median_limit_from_data(A, n_sigma=n_sigma_match)
286
287    print(f"Distance matrix: {A.shape} ({len(reference_lines)} ref × {len(peaks_measured)} detected)")
288    print(f"Tolerance: {median_limit:.2f} × median")
289
290    # Mutual NN matching
291    matches = argmin2d(A, median_limit=median_limit)
292    print(f"Mutual NN matches: {len(matches)}")
293
294    if len(matches) < 2:
295        raise RuntimeError("Not enough mutual NN matches")
296
297    y_idx = matches[:, 0]
298    x_idx = matches[:, 1]
299
300    x_matched = peaks_measured[x_idx]
301    y_matched = reference_lines[y_idx]
302
303    # Sort by x
304    idx = np.argsort(x_matched)
305    x_matched = x_matched[idx]
306    y_matched = y_matched[idx]
307
308    print(f"\nAfter matching: {len(x_matched)} pairs")
309
310    assert len(x_matched) == len(np.unique(x_matched)), "Duplicate x values!"
311    assert len(y_matched) == len(np.unique(y_matched)), "Duplicate y values!"
312
313    # --- STEP 2: OUTLIER REMOVAL ---
314    print("\n=== STEP 2: OUTLIER REMOVAL ===")
315
316    if outlier_method == 'linear':
317        x_inliers, y_inliers, _ = linear_residual_filter(
318            x_matched, y_matched, n_sigma=n_sigma_outlier
319        )
320    elif outlier_method == 'iterative':
321        x_inliers, y_inliers, _ = iterative_linear_filter(
322            x_matched, y_matched, n_sigma=n_sigma_outlier, max_iter=5
323        )
324    elif outlier_method is None:
325        print("  Skipping outlier removal")
326        x_inliers, y_inliers = x_matched, y_matched
327    else:
328        raise ValueError(f"Unknown outlier_method: {outlier_method}")
329
330    print(f"\nAfter outlier removal: {len(x_inliers)} inliers")
331
332    assert np.all(np.diff(x_inliers) > 0), "Non-monotonic x!"
333    assert np.all(np.diff(y_inliers) > 0), "Non-monotonic y!"
334
335    # --- STEP 3: INTERPOLATION ---
336    logger.info("\n=== STEP 3: INTERPOLATION ===")
337    print(f"Fitting {interpolator} on {len(x_inliers)} clean points...")
338
339    interp = get_interpolator(x_inliers, y_inliers, interpolator_method=interpolator)
340    print(interp)
341    # --- Pairs for inspection ---
342    pairs_all = np.column_stack([x_matched, y_matched])
343    pairs_inliers = np.column_stack([x_inliers, y_inliers])
344
345    return {
346        "interpolator": interp,
347        "pairs_all": pairs_all,
348        "pairs_inliers": pairs_inliers,
349    }

def robust_poly_residual_filter(x, y, deg=2, n_sigma=3.0, n_iter=300, seed=0): 
105def robust_poly_residual_filter(x, y, deg=2, n_sigma=3.0, n_iter=300, seed=0):
106    """Curve-aware robust outlier filter for matched (measured, reference) pairs.
107
108    ``linear_residual_filter`` fits a *straight line* and rejects points by their
109    residual from it. But the measured->reference relation of a Raman spectrograph is a
110    gently curved dispersion, so correctly-matched lines near the laser deviate from the
111    global straight line and get discarded. Dropping them shrinks the calibration anchor
112    span, which in turn pushes the Silicon laser-zeroing peak (and low-wavenumber sample
113    peaks) *outside* the anchor range where they are extrapolated -> the whole cm-1 axis
114    is mis-scaled and calibration worsens the spectra.
115
116    This filter keeps the largest RANSAC consensus of points lying within an adaptive band
117    of a degree-``deg`` fit, so on-trend (curved) near-laser lines survive while genuine
118    mismatches are still rejected. Returns ``(x_inliers, y_inliers, mask)`` with ``mask``
119    aligned to the input order, matching ``linear_residual_filter``.
120    """
121    x = np.asarray(x, dtype=float)
122    y = np.asarray(y, dtype=float)
123    n = len(x)
124    if n <= deg + 2:
125        return x, y, np.ones(n, dtype=bool)
126
127    xc = x - x.mean()  # centre for numerical stability of polyfit
128    # Initial (generous) acceptance band for the RANSAC search: n_sigma * robust-sigma of
129    # residuals to a curve-aware (deg) fit, floored by a fraction of the reference spacing
130    # so a near-linear region with a tiny MAD does not clip the genuinely curved tails.
131    c_all = np.polyfit(xc, y, deg)
132    r_all = y - np.polyval(c_all, xc)
133    mad = np.median(np.abs(r_all - np.median(r_all)))
134    spacing = np.median(np.abs(np.diff(np.sort(y))))
135    thr = max(n_sigma * 1.4826 * mad, 0.25 * spacing)
136    if thr <= 0:
137        thr = np.percentile(np.abs(r_all), 95)
138
139    rng = np.random.default_rng(seed)
140    best = np.zeros(n, dtype=bool)
141    for _ in range(n_iter):
142        idx = rng.choice(n, deg + 1, replace=False)
143        try:
144            c = np.polyfit(xc[idx], y[idx], deg)
145        except Exception:
146            continue
147        m = np.abs(y - np.polyval(c, xc)) < thr
148        if m.sum() > best.sum():
149            best = m
150    if best.sum() >= deg + 2:  # refit on the consensus and re-select
151        c = np.polyfit(xc[best], y[best], deg)
152        best = np.abs(y - np.polyval(c, xc)) < thr
153
154    # Doublet-misassignment split: when the matcher alternates between the components of
155    # blended reference doublets, the consensus residuals form TWO parallel populations
156    # (~a fraction of a line spacing apart). Fitting through both tilts the model; keep
157    # the majority population, which the sample-peak residuals confirm as the correct
158    # assignment (the minority would shift the axis by tens of cm-1).
159    if best.sum() >= 2 * (deg + 2):
160        c = np.polyfit(xc[best], y[best], deg)
161        rb = np.sort((y - np.polyval(c, xc))[best])
162        split_gap, split_at = 0.0, None
163        for k in range(deg + 2, len(rb) - (deg + 2) + 1):
164            a, b = rb[:k], rb[k:]
165            gap = b[0] - a[-1]
166            scatter = max(np.median(np.abs(a - np.median(a))),
167                          np.median(np.abs(b - np.median(b))), 1e-6)
168            if gap > 3.0 * 1.4826 * scatter and gap > split_gap:
169                split_gap, split_at = gap, (a[-1] + b[0]) / 2.0
170        if split_at is not None:
171            r_all_c = y - np.polyval(c, xc)
172            lower = best & (r_all_c < split_at)
173            upper = best & (r_all_c >= split_at)
174            major = lower if lower.sum() >= upper.sum() else upper
175            if major.sum() >= deg + 2:
176                print(f"  Robust deg-{deg} filter: bimodal residuals (gap={split_gap:.3f}), "
177                      f"keeping majority {major.sum()}/{best.sum()}")
178                best = major
179
180    # Tighten the band to the consensus scatter. The RANSAC band is floored by the
181    # reference line spacing, so near-miss mismatches (a fraction of a line spacing off,
182    # e.g. spurious peaks matched to the nearest available line) survive the search and
183    # tilt the subsequent model fit -- the exact "slope error amplified by Si zeroing"
184    # failure. The consensus residual MAD is the real measurement noise; rescale to it,
185    # with a small floor so curved tails beyond the deg-fit are not clipped.
186    if best.sum() >= deg + 2:
187        c = np.polyfit(xc[best], y[best], deg)
188        r = y - np.polyval(c, xc)
189        mad_c = np.median(np.abs(r[best] - np.median(r[best])))
190        thr2 = min(thr, max(n_sigma * 1.4826 * mad_c, 0.05 * spacing))
191        tight = np.abs(r) < thr2
192        if tight.sum() >= deg + 2:
193            c = np.polyfit(xc[tight], y[tight], deg)  # final refit on the tightened set
194            tight = np.abs(y - np.polyval(c, xc)) < thr2
195            if tight.sum() >= deg + 2:
196                best, thr = tight, thr2
197
198    print(f"  Robust deg-{deg} filter: band={thr:.3f}, kept {best.sum()}/{n} inliers")
199    return x[best], y[best], best

def iterative_linear_filter(x, y, n_sigma=3.0, max_iter=5): 
202def iterative_linear_filter(x, y, n_sigma=3.0, max_iter=5):
203    """
204    Iteratively fit linear model and remove outliers.
205
206    Args:
207        x, y: Matched pairs (normalized)
208        n_sigma: Number of MAD sigmas
209        max_iter: Maximum iterations
210
211    Returns:
212        x_inliers, y_inliers, mask
213    """
214    mask = np.ones(len(x), dtype=bool)
215
216    for iteration in range(max_iter):
217        x_fit = x[mask]
218        y_fit = y[mask]
219
220        if len(x_fit) < 2:
221            print(f"  Iteration {iteration}: Too few points, stopping")
222            break
223
224        # Fit linear model on current inliers
225        a, b = np.polyfit(x_fit, y_fit, deg=1)
226
227        # Compute residuals on ALL points
228        y_pred = a * x + b
229        resid = y - y_pred
230
231        # MAD on current inliers
232        resid_inliers = resid[mask]
233        med = np.median(resid_inliers)
234        mad = np.median(np.abs(resid_inliers - med))
235
236        if mad < 1e-10:
237            print(f"  Iteration {iteration}: Converged (MAD={mad:.2e})")
238            break
239
240        threshold = n_sigma * mad
241        new_mask = np.abs(resid - med) < threshold
242
243        n_removed = (mask & ~new_mask).sum()
244        print(f"  Iteration {iteration}: {new_mask.sum()}/{len(x)} inliers, "
245              f"removed {n_removed}, MAD={mad:.6f}")
246
247        if np.array_equal(mask, new_mask):
248            print("  Converged (mask unchanged)")
249            break
250
251        mask = new_mask
252
253    return x[mask], y[mask], mask

521    # Mark target reference if present
522    if target_ref in reference_lines or any(np.abs(reference_lines - target_ref) < 0.5):
523        if idx_target < m_show:
524            ax2.axhline(idx_target, color='red', linewidth=2, alpha=0.7)

526                     color='red', va='center', fontweight='bold')
527

529    plt.savefig(output_file, dpi=150, bbox_inches='tight')

531
532    return fig
