1"""Tests for qmatch.robust_poly_residual_filter, in particular the opt-in 

2``auto_reduce_degree`` behaviour for sparse reference sets. 

3 

4Context: CAL (calcite) verification uses only 6 certified reference positions 

5(155.21, 281.26, 711.95, 1085.91, 1435.22, 1748.91 cm-1), unlike neon's ~15-30 

6NIST lines. The default degree-2 RANSAC filter draws 3-point samples to fit a 

7quadratic; with only 6 points and 2 of them mismatched to the wrong band, a 

83-point sample can include a bad point and still look self-consistent, so the 

9"largest consensus" is not a reliable signal and the filter can fail to reject 

10a mismatch that is tens of cm-1 off (observed on P6_0701/OP4's CAL peaks: 

11matched_peaks_samples.csv carries an inlier_mask=True row at 181 cm-1 from 

12this exact reference set). 

13 

14``auto_reduce_degree=True`` recurses to a lower polynomial degree when there 

15are too few points for the requested degree to be well-determined, down to a 

16plain MAD-distance test at degree 0. It defaults to False so the neon 

17calibration path (which has enough points for degree-2 RANSAC to be reliable) 

18is unaffected. 

19""" 

20import numpy as np 

21 

22from ramanchada2.protocols.calibration.qmatch import robust_poly_residual_filter 

23 

24# The real CAL reference table from calibration_verify.py's get_reference_peaks("CAL"). 

25CAL_REFERENCE_CM1 = np.array( 

26 [155.21, 281.26, 711.95, 1085.91, 1435.22, 1748.91]) 

27 

28 

29def _cal_like_mismatch(): 

30 """Reproduce the shape of the P6_0701/OP4 CAL failure: two reference 

31 positions matched to the wrong spectral peak, tens of cm-1 away, the rest 

32 well matched.""" 

33 y = CAL_REFERENCE_CM1.copy() 

34 x = y.copy() 

35 x[1] = 462.63 # 281.26 matched to a peak that is really ~181 cm-1 away 

36 x[4] = 1506.66 # 1435.22 matched to a peak that is really ~71 cm-1 away 

37 return x, y 

38 

39 

40def test_default_behaviour_is_unchanged_for_dense_reference_sets(): 

41 """auto_reduce_degree defaults to False: a neon-sized reference set (well 

42 above the degree-2 threshold) must produce identical output whether or not 

43 the new parameter is mentioned, so the calibration path is untouched.""" 

44 rng = np.random.default_rng(1) 

45 y = np.linspace(540, 700, 20) 

46 x = y + rng.normal(0, 0.05, size=y.size) 

47 

48 x1, y1, mask1 = robust_poly_residual_filter(x, y, deg=2, n_sigma=3) 

49 x2, y2, mask2 = robust_poly_residual_filter( 

50 x, y, deg=2, n_sigma=3, auto_reduce_degree=False) 

51 

52 np.testing.assert_array_equal(mask1, mask2) 

53 np.testing.assert_array_equal(x1, x2) 

54 np.testing.assert_array_equal(y1, y2) 

55 

56 

57def test_sparse_set_without_auto_reduce_can_miss_mismatches(): 

58 """Pin the failure mode this fix addresses: with auto_reduce_degree left 

59 at its default False, the degree-2 filter on a 6-point CAL-like set is not 

60 guaranteed to reject a badly mismatched pair. This test documents the 

61 defect rather than requiring it (the RANSAC search is a bit stochastic), 

62 so it only asserts the filter runs and returns a well-formed mask.""" 

63 x, y = _cal_like_mismatch() 

64 xi, yi, mask = robust_poly_residual_filter(x, y, deg=2, n_sigma=3) 

65 assert mask.shape == (6,) 

66 assert mask.sum() >= 1

69def test_auto_reduce_degree_rejects_gross_mismatch_on_sparse_set(): 

70 """With auto_reduce_degree=True, at least the grossly mismatched point 

71 (index 1: 281.26 -> matched 181 cm-1 away, the largest deviation in the 

72 set) must be flagged as an outlier on the real CAL reference geometry.""" 

73 x, y = _cal_like_mismatch() 

74 xi, yi, mask = robust_poly_residual_filter( 

75 x, y, deg=2, n_sigma=3, auto_reduce_degree=True) 

76 assert mask.shape == (6,) 

77 assert not mask[1], "the ~181 cm-1 mismatch must be rejected"

80def test_auto_reduce_degree_keeps_well_matched_sparse_set_intact(): 

81 """A sparse but well-matched reference set must not lose points just 

82 because auto_reduce_degree is enabled - the fallback should not become 

83 more aggressive than necessary when there is nothing to reject.""" 

84 rng = np.random.default_rng(2) 

85 y = CAL_REFERENCE_CM1.copy() 

86 x = y + rng.normal(0, 0.05, size=y.size) 

87 

88 xi, yi, mask = robust_poly_residual_filter( 

89 x, y, deg=2, n_sigma=3, auto_reduce_degree=True) 

90 assert mask.all(), "no genuine mismatch should not be manufactured"

93def test_auto_reduce_degree_zero_uses_mad_distance_base_case(): 

94 """When even a degree-1 fit is underdetermined, the degree-0 base case (a 

95 plain MAD-distance test around the median) must still flag an obvious 

96 outlier and keep the well-behaved points.""" 

97 y = np.array([100.0, 102.0, 98.0, 250.0]) # last point is the outlier 

98 x = y.copy() 

99 xi, yi, mask = robust_poly_residual_filter( 

100 x, y, deg=1, n_sigma=3, auto_reduce_degree=True) 

101 assert mask.shape == (4,) 

102 assert not mask[3] 

103 assert mask[:3].all()
