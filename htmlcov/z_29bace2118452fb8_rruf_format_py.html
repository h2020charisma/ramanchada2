2from typing import Dict, List, Tuple

4import pandas 

5from numpy.typing import NDArray

7# A real data row is "<number><sep><number>" (RRUFF's x, y body). Real 

8# RRUFF exports interleave two kinds of non-"##" lines before the actual 

9# data starts: a blank separator line, and continuation text wrapped from 

10# the preceding "##KEY=..." value onto a line that itself has no "##" 

11# prefix (e.g. a multi-line ##ORIENTATION note). Neither is data, so 

12# "first line not starting with ##" (the original check) lands on 

13# whichever of those comes first instead of the real first data row -- 

14# confirmed against real downloaded rruff.net corpus files, where this 

15# was true for the large majority (blank-line case) and a smaller but 

16# non-trivial fraction (wrapped-continuation case). "RAW" files (as 

17# opposed to "Processed") additionally use scientific notation on BOTH 

18# columns (e.g. "8.758203e+001, 5.919380e+002"), so the position column 

19# needs the same e/+/- characters as the intensity column, not just 

20# digits and a decimal point. A minority of RAW files use a tab, or (rarer 

21# still) plain whitespace, instead of a comma as the column separator -- 

22# also confirmed against the real corpus -- so all three are accepted. 

23_NUMBER = r'-?[0-9.]+(?:[eE][+-]?[0-9]+)?' 

24_SEP = r'(?:,|\t|[ \t]+)' 

25_DATA_ROW = re.compile(r'^\s*{n}\s*{sep}\s*{n}\s*$'.format(n=_NUMBER, sep=_SEP)) 

26# One split per row: a comma (with any surrounding whitespace collapsed 

27# into it) or a run of whitespace -- NOT "comma or whitespace" matched 

28# independently, which would treat "1.0, 2.0"'s comma AND its following 

29# space as two separate splits and produce 3 fields instead of 2. 

30_SPLIT_SEP = re.compile(r'\s*,\s*|\s+') 

31 

32 

33def rruf_format(lines: List[str]) -> Tuple[NDArray, NDArray, Dict]: 

34 header_lines = [] 

35 start_spe = None 

36 for i, ll in enumerate(lines): 

37 if ll.startswith('##'): 

38 header_lines.append(ll) 

39 elif _DATA_ROW.match(ll): 

40 start_spe = i 

41 break 

42 # else: blank separator or wrapped header continuation -- skip, 

43 # keep looking for the first genuine data row. 

44 if start_spe is None: 

45 raise ValueError('rruf_format: no numeric data rows found') 

46 meta = dict([ll.strip()[2:].split('=', 1) for ll in header_lines]) 

47 stop_spe = len(lines) 

48 for i, ll in enumerate(lines): 

49 if ll.startswith('##END'): 

50 stop_spe = i 

51 break 

52 data = pandas.DataFrame.from_records( 

53 data=[_SPLIT_SEP.split(ll.strip()) for ll in lines[start_spe:stop_spe] if ll.strip()] 

54 ).apply(pandas.to_numeric).dropna(axis=0) 

55 positions, intensities = data.to_numpy().T 

56 return positions, intensities, meta
