1"""CWA 18133:2024 §8 portable calibration files. 

2 

3§8 asks for a calibration file with metadata, date, the calibration curve as points 

4(uncalibrated shift -> calibrated shift), the silicon peak position and (optionally) the 

5calibrated laser wavelength; the curve points are meant to regenerate a spline. These 

6exporters write that as ``<base>.csv`` (the curve) + ``<base>.json`` (metadata AND the full 

7model in the portable :meth:`CalibrationModel.to_dict` form, so the exact model — not just 

8the sampled curve — can be reconstructed by ramanchada2 or any JSON-capable reader). 

9 

10``export_nexus_calibration`` writes the same content, plus the calibrant spectra it was 

11derived from, as a self-describing NXraman HDF5 file (plain h5py or h5pyd). 

12"""

33def _calibration_curve(calmodel, spectral_range, npoints): 

34 """Sample the model over ``spectral_range`` (cm-1): uncalibrated -> calibrated.""" 

35 lo, hi = float(min(spectral_range)), float(max(spectral_range)) 

36 grid = np.linspace(lo, hi, int(npoints)) # endpoints exactly at the range bounds (§8) 

37 spe = Spectrum(x=grid, y=np.ones_like(grid)) 

38 calibrated = calmodel.apply_calibration_x(spe, spe_units="cm-1").x 

39 return grid, np.asarray(calibrated, dtype=float)

42def _laser_zero_info(calmodel): 

43 """Si peak position (nm and implied cm-1 on the nominal axis) and calibrated laser wl.""" 

44 from .xcalibration import LazerZeroingComponent 

45 for c in calmodel.components: 

46 if isinstance(c, LazerZeroingComponent): 

47 zero_nm = float(c.model) 

48 si_ref = float(list(c.ref.keys())[0]) if c.ref else SI_REF_CM1 

49 laser_nm = 1e7 / (1e7 / zero_nm + si_ref) 

50 return { 

51 "si_peak_nm": zero_nm, 

52 "si_reference_cm1": si_ref, 

53 "calibrated_laser_wl_nm": laser_nm, 

54 } 

55 return {}

58def export_cwa_x(calmodel, base_path, spectral_range, npoints=200, metadata=None): 

59 """Write ``<base_path>.csv`` + ``<base_path>.json`` for an x-calibration model. 

60 

61 Args: 

62 calmodel: a derived :class:`CalibrationModel` (Ne curve + Si zeroing components). 

63 base_path: output path without extension. 

64 spectral_range: (min, max) Raman shift in cm-1 of the spectra this calibration is 

65 intended for; the CSV endpoints correspond exactly to these bounds (§8). 

66 npoints: number of curve points. 

67 metadata: optional dict merged into the JSON (instrument metadata per CWA §5.3.3). 

68 """ 

69 grid, calibrated = _calibration_curve(calmodel, spectral_range, npoints) 

70 csv_path = f"{base_path}.csv" 

71 json_path = f"{base_path}.json" 

72 with open(csv_path, "w", encoding="utf-8") as f: 

73 f.write("uncalibrated_cm1,calibrated_cm1\n") 

74 for u, c in zip(grid, calibrated): 

75 f.write(f"{u:.6f},{c:.6f}\n") 

76 

77 doc = { 

78 "format": "CWA18133-x-calibration", 

79 "date": datetime.datetime.now().isoformat(timespec="seconds"), 

80 "laser_wl_nominal_nm": int(calmodel.laser_wl) if calmodel.laser_wl is not None else None, 

81 **_laser_zero_info(calmodel), 

82 "spectral_range_cm1": [float(min(spectral_range)), float(max(spectral_range))], 

83 "curve_csv": csv_path.replace("\\", "/").split("/")[-1], 

84 "metadata": metadata or {}, 

85 "model": calmodel.to_dict(), 

86 } 

87 with open(json_path, "w", encoding="utf-8") as f: 

88 json.dump(doc, f, indent=1) 

89 return csv_path, json_path

92def export_cwa_y(ycal_component, base_path, spectral_range=None, npoints=200, 

93 metadata=None, x_calibration_ref=None): 

94 """Write ``<base_path>.csv`` + ``<base_path>.json`` for a y-calibration component. 

95 

96 The CSV holds intensity factors as a function of the calibrated Raman shift (§8): 

97 factor(x) = certificate_response(x) / measured_reference(x), masked where the measured 

98 reference is at/below its noise (same rule as YCalibrationComponent.safe_factor). 

99 """ 

100 cert = ycal_component.ref 

101 if spectral_range is None: 

102 spectral_range = cert.raman_shift 

103 lo, hi = float(min(spectral_range)), float(max(spectral_range)) 

104 grid = np.linspace(lo, hi, int(npoints)) 

105 measured = np.asarray(ycal_component.model(grid), dtype=float) 

106 expected = np.asarray(cert.Y(grid), dtype=float) 

107 factor = np.zeros_like(grid) 

108 ref_noise = 0.0 # dense evaluation of the smoothed model; mask only non-positive values 

109 mask = (measured > ref_noise) & np.isfinite(expected) 

110 factor[mask] = expected[mask] / measured[mask] 

111 

112 csv_path = f"{base_path}.csv" 

113 json_path = f"{base_path}.json" 

114 with open(csv_path, "w", encoding="utf-8") as f: 

115 f.write("calibrated_cm1,intensity_factor\n") 

116 for x, y in zip(grid, factor): 

117 f.write(f"{x:.6f},{y:.8g}\n") 

118 

119 doc = { 

120 "format": "CWA18133-y-calibration", 

121 "date": datetime.datetime.now().isoformat(timespec="seconds"), 

122 "laser_wl_nominal_nm": int(ycal_component.laser_wl) if ycal_component.laser_wl is not None else None, 

123 "certificate": cert.model_dump(), 

124 "x_calibration_ref": x_calibration_ref, 

125 "spectral_range_cm1": [lo, hi], 

126 "curve_csv": csv_path.replace("\\", "/").split("/")[-1], 

127 "metadata": metadata or {}, 

128 "model": ycal_component.to_dict(), 

129 } 

130 with open(json_path, "w", encoding="utf-8") as f: 

131 json.dump(doc, f, indent=1) 

132 return csv_path, json_path

142def _is_missing(value): 

143 """True for scalars that must not be written: None, pd.NA/NaT, NaN of any float width.""" 

144 if value is None or value is pd.NA or value is pd.NaT: 

145 return True 

146 if isinstance(value, (float, np.floating)): 

147 return bool(np.isnan(value)) 

148 if isinstance(value, np.datetime64): 

149 return bool(np.isnat(value)) 

150 return False

154 """json.dumps ``default=`` hook for numpy / pandas / datetime values.""" 

155 if _is_missing(obj): 

156 return None 

157 if isinstance(obj, np.ndarray): 

158 return obj.tolist() 

159 if isinstance(obj, np.generic):

166def _to_h5_value(value, h5): 

167 """Coerce an arbitrary metadata/to_dict() value into something ``create_dataset`` 

168 accepts, or return ``None`` if it is missing and must be dropped. 

169 

170 Scalars: numpy strings/ints/floats/bools become plain python scalars; pd.NA/NaT/NaN 

171 (any float width) are dropped; datetimes (pd.Timestamp included) become ISO-8601 

172 strings; bytes are decoded. Sequences/arrays: all-string -> variable-length utf-8 

173 string array, numeric -> numeric array, anything else (mixed, ragged, nested objects, 

174 dicts) -> a JSON string. Unknown objects fall back to ``str``. 

175 """ 

176 if _is_missing(value): 

177 return None 

178 if isinstance(value, (bool, np.bool_)): 

179 return bool(value) 

180 if isinstance(value, (str, np.str_)): 

181 return str(value) 

182 if isinstance(value, bytes): 

183 return value.decode("utf-8", errors="replace") 

184 if isinstance(value, np.integer): 

185 return int(value) 

186 if isinstance(value, (int, float)): 

187 return value 

188 if isinstance(value, np.floating): 

189 return float(value) 

190 if isinstance(value, np.datetime64): 

191 return str(np.datetime_as_string(value)) 

192 if isinstance(value, (datetime.datetime, datetime.date)): 

193 return value.isoformat() 

194 if isinstance(value, (pd.Timedelta, np.timedelta64, datetime.timedelta)): 

195 return str(value) 

196 if isinstance(value, dict): 

197 return json.dumps(value, default=_json_default) 

198 if isinstance(value, (list, tuple, np.ndarray, pd.Series)): 

199 if isinstance(value, np.ndarray) and value.ndim == 0: 

200 return _to_h5_value(value.item(), h5) 

201 seq = value.tolist() if isinstance(value, (np.ndarray, pd.Series)) else list(value) 

202 if seq and all(isinstance(v, (str, np.str_)) for v in seq): 

203 return np.asarray([str(v) for v in seq], dtype=_string_dtype(h5)) 

204 try: 

205 arr = np.asarray(seq) 

206 except ValueError: # ragged 

207 arr = None 

208 if arr is not None and arr.dtype.kind in "biuf": 

209 return arr 

210 return json.dumps(seq, default=_json_default) 

211 return str(value)

219def _clean_mapping(mapping, h5): 

220 """``{safe_key: h5-ready value}`` for ``mapping``: missing values dropped, '/' in keys 

221 replaced (a '/' would silently create nested HDF5 groups), clashes disambiguated.""" 

222 out = {} 

223 for key, value in (mapping or {}).items(): 

224 cleaned = _to_h5_value(value, h5) 

225 if cleaned is None: 

226 continue 

227 safe = str(key).replace("/", "_").strip() or "_" 

228 base, i = safe, 1 

229 while safe in out: 

230 i += 1 

231 safe = f"{base}_{i}" 

232 out[safe] = cleaned 

233 return out

236def _component_parameters(component_dict): 

237 """Extract the reconstructable fit parameters from one to_dict() component payload, 

238 for NXcalibration/calibration_parameters — separate from the anchors (matched peaks, 

239 a different length) and from calibration_object (the full JSON, kept verbatim). 

240 

241 Returns a flat ``{name: value}`` dict of JSON-scalar/array values suitable for writing 

242 as sibling datasets in an NXparameters group. Shape depends on the interpolator/ 

243 component type; unknown shapes fall back to an empty dict rather than raising, since 

244 calibration_object already carries the complete, authoritative payload. 

245 """ 

246 model = component_dict.get("model") 

247 out = {} 

248 if component_dict.get("type") == "LazerZeroingComponent": 

249 # model is a plain float: the fitted Si peak position, nm 

250 out["si_peak_nm"] = float(model) 

251 return out 

252 if not isinstance(model, dict): 

253 return out 

254 kind = model.get("type") 

255 if kind == "CustomPolyInterpolator": 

256 for key in ("coef", "degree", "x_min", "x_max", "fit_error"): 

257 if key in model and model[key] is not None: 

258 out[key] = model[key] 

259 elif kind in ("CustomPChipInterpolator", "CustomCubicSplineInterpolator"): 

260 if "x" in model: 

261 out["knots_x"] = model["x"] 

262 if "y" in model: 

263 out["knots_y"] = model["y"] 

264 elif kind == "ParametricModel": 

265 out["equation"] = model.get("equation") 

266 out["param_names"] = model.get("param_names") 

267 out["coef"] = model.get("coef") 

268 return out

271def _fit_formula_description(component_dict): 

272 """Human-readable summary of the interpolator + its extrapolation rule. 

273 

274 Both CustomPolyInterpolator and CustomPChipInterpolator continue with CONSTANT 

275 CORRECTION (unit slope) beyond the anchor span rather than an unconstrained tail 

276 (see interpolators.py) — a reader that reimplements naive extrapolation will diverge 

277 from this model by tens of nm outside the anchor range, so the rule is stated here. 

278 """ 

279 if component_dict.get("type") == "LazerZeroingComponent": 

280 profile = component_dict.get("profile", "Pearson4") 

281 return f"Silicon 520.45 cm-1 peak position, fitted with a {profile} profile." 

282 model = component_dict.get("model") 

283 kind = model.get("type") if isinstance(model, dict) else None 

284 extrapolation = ("Beyond the anchor span the map continues with constant correction " 

285 "(unit slope) from the edge value, not raw polynomial/spline " 

286 "extrapolation.") 

287 if kind == "CustomPolyInterpolator": 

288 degree = model.get("degree") 

289 return (f"Polynomial degree {degree}, fit on x normalized to " 

290 f"u=(x-x_min)/(x_max-x_min); calibrated = polyval(coef, u). " 

291 f"{extrapolation}") 

292 if kind == "CustomPChipInterpolator": 

293 return f"Monotone PCHIP through (knots_x, knots_y). {extrapolation}" 

294 if kind == "CustomCubicSplineInterpolator": 

295 bc = model.get("bc_type") if isinstance(model, dict) else None 

296 return f"Cubic spline (bc_type={bc}) through (knots_x, knots_y). {extrapolation}" 

297 if kind == "ParametricModel": 

298 return f"Analytic model: {model.get('equation')}" 

299 return kind or "unknown"

340@contextlib.contextmanager 

341def _create_h5_file(h5, filename, atomic): 

342 """Open ``filename`` for writing; with ``atomic`` write ``<filename>.tmp`` and 

343 ``os.replace`` it into place only if the body succeeded.""" 

344 if not atomic: 

345 with h5.File(filename, "w") as f: 

346 yield f 

347 return

350 with h5.File(tmp, "w") as f:

353 except BaseException:

355 os.remove(tmp)

386def _nx_calibration_group(parent, name, component_dict, h5, curve=None, curve_units=None): 

387 """Write one calibration component as an NXcalibration group under ``parent``. 

388 

389 ``curve`` is an optional (original_axis, calibrated_axis) array pair — real datasets — 

390 and ``curve_units`` the matching (original, calibrated) ramanchada2 unit strings. 

391 calibration_parameters is an NXparameters container with the actual numeric children; 

392 calibration_object/NXnote is kept deliberately small — only fields with no typed home 

393 yet (see _residual_component_fields) — so the typed datasets are the single source of 

394 truth for the reconstructable numbers rather than a duplicate copy. 

395 """ 

396 is_y = component_dict.get("type") == "YCalibrationComponent" 

397 cal = _nx_group( 

398 parent, name, "NXcalibration", 

399 description="ramanchada2 " + component_dict.get("type", "calibration"), 

400 physical_quantity="relative intensity" if is_y else "wavenumber", 

401 applied=bool(component_dict.get("enabled", True)), 

402 fit_formula_description=_fit_formula_description(component_dict), 

403 ) 

404 

405 if curve is not None: 

406 original_axis, calibrated_axis = curve 

407 orig_ds = cal.create_dataset("original_axis", data=np.asarray(original_axis, dtype=float)) 

408 cal_ds = cal.create_dataset("calibrated_axis", data=np.asarray(calibrated_axis, dtype=float)) 

409 if curve_units: 

410 orig_ds.attrs["units"] = _nx_units(curve_units[0]) 

411 cal_ds.attrs["units"] = _nx_units(curve_units[1]) 

412 

413 params = _component_parameters(component_dict) 

414 if params: 

415 pgroup = _nx_group(cal, "calibration_parameters", "NXparameters") 

416 for key, value in _clean_mapping(params, h5).items(): 

417 pgroup.create_dataset(key, data=value) 

418 

419 _nx_note(cal, "calibration_object", _residual_component_fields(component_dict), h5) 

420 

421 anchors = component_dict.get("anchors") 

422 if anchors: 

423 arr = np.asarray(anchors, dtype=float) 

424 agroup = _nx_group( 

425 cal, "anchors", "NXdata", 

426 description="Matched calibrant peaks used to derive the model.") 

427 agroup.create_dataset("measured", data=arr[:, 0]) 

428 agroup.create_dataset("reference", data=arr[:, 1]) 

429 agroup.create_dataset("inlier", data=arr[:, 2].astype(bool)) 

430 return cal

433def _component_curves(calmodel, grid, spe_units="cm-1"): 

434 """Per-stage x curves, mirroring :meth:`CalibrationModel.apply_calibration_x`. 

435 

436 Returns ``{id(component): (original_x, calibrated_x, original_units, calibrated_units)}`` 

437 for each *enabled* component: its own input axis (the previous stage's output) and its 

438 own output axis, in the units each stage works in (e.g. the Ne component maps cm-1 to 

439 nm, the Si zeroing component maps nm back to cm-1). Components whose output length 

440 differs from the input (non-monotonic 'drop' policy) get no entry. 

441 """ 

442 from .xcalibration import LazerZeroingComponent 

443 curves = {} 

444 x = np.array(grid, dtype=float) 

445 axis_units = spe_units # units of the axis actually carried between stages 

446 process_units = spe_units # what apply_calibration_x passes as spe_units to the stage 

447 for comp in getattr(calmodel, "components", []): 

448 if not comp.enabled: 

449 continue 

450 in_x = np.array(x, dtype=float) 

451 # fresh Spectrum per stage: process() may assign to .x of the spectrum it is given 

452 out = comp.process(Spectrum(x=in_x.copy(), y=np.ones_like(in_x)), process_units, 

453 convert_back=False) 

454 out_x = np.array(out.x, dtype=float) 

455 # LazerZeroingComponent.model_units is its *input* unit (nm); it outputs cm-1 

456 out_units = "cm-1" if isinstance(comp, LazerZeroingComponent) else comp.model_units 

457 if out_x.shape == in_x.shape: 

458 curves[id(comp)] = (in_x, out_x, axis_units, out_units) 

459 x, axis_units, process_units = out_x, out_units, comp.model_units 

460 return curves

463def _y_curves(ycal_component, grid, npoints): 

464 """Sample the y calibration on the certificate's declared range. 

465 

466 Returns ``(x, measured, certificate, factor)``: ``measured`` is the measured-SRM 

467 response model (``ycal_component.model``), ``certificate`` the certified response and 

468 ``factor`` the intensity factor ``YCalibrationComponent.process`` multiplies spectra by 

469 (certificate / measured, 0 where the measured response is not usable). ``factor`` is 

470 obtained by running ``process`` on a unit spectrum, so it is by construction what the 

471 component applies. 

472 """ 

473 y_grid = np.asarray(grid, dtype=float) 

474 cert = getattr(ycal_component, "ref", None) 

475 if cert is not None and getattr(cert, "raman_shift", None): 

476 lo, hi = cert.raman_shift 

477 y_grid = np.linspace(float(lo), float(hi), int(npoints)) 

478 measured = np.asarray(ycal_component.model(y_grid), dtype=float) 

479 certificate = _certificate_curve(cert, y_grid) if cert is not None else None 

480 factor = np.asarray( 

481 ycal_component.process(Spectrum(x=y_grid.copy(), y=np.ones_like(y_grid))).y, 

482 dtype=float) 

483 return y_grid, measured, certificate, factor

486def _beam_wavelength(wavelength, instrument, calmodel): 

487 """Beam wavelength in nm: explicit argument, else instrument['laser_wl'], else 

488 ``calmodel.laser_wl``. Returns ``(wavelength_or_None, instrument_laser_wl_consumed)``; 

489 the second item is True when ``instrument['laser_wl']`` only repeats the beam wavelength 

490 (so it need not also be written as a parameter). An explicit wavelength that differs 

491 from ``instrument['laser_wl']`` leaves the latter in place.""" 

492 if wavelength is not None: 

493 value = float(wavelength) 

494 try: 

495 same = float(instrument.get("laser_wl")) == value 

496 except (TypeError, ValueError): 

497 same = False 

498 return value, same 

499 for source, candidate in (("instrument", instrument.get("laser_wl")), 

500 ("calmodel", getattr(calmodel, "laser_wl", None))): 

501 try: 

502 value = float(candidate) 

503 except (TypeError, ValueError): 

504 continue 

505 if np.isfinite(value): 

506 return value, source == "instrument" 

507 return None, False

515 metadata=None, 

516 instrument=None, 

517 ycal_component=None, 

518 spe_neon=None, 

519 spe_neon_units="cm-1", 

520 spe_silicon=None, 

521 spe_silicon_units="cm-1", 

522 title=None, 

523 wavelength=None, 

524 vendor=None, 

525 model=None, 

526 h5module=None, 

527): 

528 """Write the calibration workflow as a self-describing NeXus (NXraman) HDF5 file. 

529 

530 Only ``h5py`` (or ``h5pyd`` via ``h5module``, as in :func:`ramanchada2.io.HSDS.write_nexus`) 

531 is used. The entry carries ``definition = "NXraman"``, an ``NXinstrument`` (device 

532 information, incident beam wavelength, one ``NXcalibration`` group per calibration 

533 component), an ``NXsample`` and plottable ``NXdata`` groups for the calibrant spectra 

534 actually used (``spe_neon``/``spe_silicon``, as loaded), the sampled x calibration curve 

535 and, with ``ycal_component``, the y calibration curves. 

536 

537 The full model is stored as JSON in ``entry/calibration_model/data``, so a reader can 

538 regenerate ``calmodel`` from the file alone via 

539 ``CalibrationModel.from_dict(json.loads(...)["model"])``. 

540 

541 x curves. ``entry/calibration_curve`` is the end-to-end map (uncalibrated cm-1 -> 

542 calibrated cm-1). Each ``instrument/calibration_x_<i>`` group carries *its own stage's* 

543 ``original_axis`` -> ``calibrated_axis`` pair, which is generally in different units 

544 (the Ne component maps cm-1 -> nm, the Si laser-zeroing component maps nm -> cm-1); 

545 the datasets' ``units`` attrs state which. Chaining the stages reproduces 

546 ``calibration_curve``. 

547 

548 y curves (on the calibrated Raman shift axis, over the certificate's declared range). 

549 ``calibration_curve_y`` (signal ``intensity_factor``) is the factor 

550 ``YCalibrationComponent.process`` multiplies a spectrum by, i.e. certificate response / 

551 measured response (0 where the measured response is not usable). 

552 ``calibration_curve_y_measured`` (signal ``measured_response``) and 

553 ``calibration_curve_y_certificate`` (signal ``certificate_response``) are the two 

554 responses it is formed from. The measured response is the model fitted to the measured 

555 SRM spectrum (normalised unless the component was built with ``normalize=False``), so 

556 only the factor, not the individual response scales, is meaningful in absolute terms. 

557 If the y curves cannot be evaluated a warning is logged and ``calibration_y`` gets a 

558 ``curves_error`` attribute; the curve groups are then absent. 

559 

560 Args: 

561 calmodel: a derived :class:`CalibrationModel` (Ne curve + Si zeroing components). 

562 filename: output ``.nxs``/``.h5`` path (or HSDS domain for ``h5pyd``). 

563 spectral_range: (min, max) Raman shift in cm-1 the sampled curve should cover. 

564 npoints: number of curve points. 

565 metadata: optional dict stored in the JSON model document. 

566 instrument: optional dict of instrument metadata. ``instrument_make``/ 

567 ``instrument_model`` become ``NXinstrument/device_information`` (vendor, 

568 model); ``laser_wl`` is used for the beam wavelength if ``wavelength`` is not 

569 given (falling back to ``calmodel.laser_wl``); other keys are written as 

570 datasets under ``instrument/parameters``. None/NaN/pd.NA/NaT values are 

571 dropped, numpy and pandas scalars are converted (Timestamps to ISO-8601 

572 strings), sequences become arrays (or a JSON string if not homogeneous) and 

573 '/' in keys is replaced by '_'. 

574 ycal_component: optional derived ``YCalibrationComponent`` (relative-intensity 

575 calibration); written as a second NXcalibration group together with its 

576 y curves (see above). 

577 spe_neon, spe_silicon: optional as-loaded calibrant :class:`Spectrum` objects (the 

578 *inputs* the model was derived from). When omitted, no reference spectra are 

579 written. 

580 spe_neon_units, spe_silicon_units: axis units of the calibrant spectra as loaded 

581 ("cm-1", "nm", or "pixel"); do not assume cm-1. Written as NeXus unit strings 

582 ("1/cm", "nm", "pixel"). 

583 title: optional entry title. 

584 wavelength: incident laser wavelength in nm. 

585 vendor, model: instrument vendor and model for ``NXinstrument/device_information``. 

586 They take precedence over ``instrument["instrument_make"]`` / 

587 ``instrument["instrument_model"]``; when neither is given the field is omitted 

588 (nothing is invented). 

589 h5module: ``h5py`` (default) or ``h5pyd``. 

590 

591 Returns: 

592 filename 

593 """ 

594 # a local file (default h5py) is written to a temp name and moved into place on success, 

595 # so a failure never leaves a truncated file or destroys an existing one; an h5pyd 

596 # domain is not a local path and is written directly 

597 atomic = h5module is None 

598 if h5module is None: 

599 import h5py as h5module 

600 h5 = h5module 

601 

602 grid, calibrated = _calibration_curve(calmodel, spectral_range, npoints) 

603 stage_curves = _component_curves(calmodel, grid) 

604 

605 instrument = _clean_mapping(instrument, h5) 

606 wavelength, laser_wl_consumed = _beam_wavelength(wavelength, instrument, calmodel) 

607 if laser_wl_consumed: 

608 instrument.pop("laser_wl") # numeric: it is the beam wavelength, not a parameter 

609 

610 spectra = [] # (group name, x, y, x_units, y_name, description) 

611 for name, spe, units in (("reference_neon", spe_neon, spe_neon_units), 

612 ("reference_silicon", spe_silicon, spe_silicon_units)): 

613 if spe is not None: 

614 spectra.append((name, spe.x, spe.y, units, "intensity", "As-loaded calibrant.")) 

615 

616 y_data = None 

617 y_error = None 

618 if ycal_component is not None: 

619 try: 

620 y_data = _y_curves(ycal_component, grid, npoints) 

621 except (ValueError, ArithmeticError) as err: 

622 y_error = f"{type(err).__name__}: {err}" 

623 logger.warning("NeXus export: y calibration curves not written (%s)", y_error) 

624 

625 with _create_h5_file(h5, filename, atomic) as f: 

626 f.attrs["NX_class"] = "NXroot" 

627 f.attrs["default"] = "entry" 

628 entry = _nx_group(f, "entry", "NXentry", default="calibration_curve") 

629 entry.create_dataset("definition", data="NXraman") 

630 entry.create_dataset("title", data=title or "ramanchada2 x/y calibration") 

631 entry.create_dataset("experiment_type", data="Raman spectroscopy") 

632 

633 inst = _nx_group(entry, "instrument", "NXinstrument") 

634 device_fields = {"vendor": vendor or instrument.get("instrument_make"), 

635 "model": model or instrument.get("instrument_model")} 

636 device_fields = {k: str(v) for k, v in device_fields.items() if v} 

637 if device_fields: 

638 device = _nx_group(inst, "device_information", "NXfabrication") 

639 for key, value in device_fields.items(): 

640 device.create_dataset(key, data=value) 

641 if wavelength is not None: 

642 beam = _nx_group(inst, "beam_incident", "NXbeam") 

643 wl = beam.create_dataset("wavelength", data=float(wavelength)) 

644 wl.attrs["units"] = "nm" 

645 extras = {k: v for k, v in instrument.items() 

646 if k not in ("instrument_make", "instrument_model")} 

647 if extras: 

648 params = _nx_group(inst, "parameters", "NXparameters") 

649 for key, value in extras.items(): 

650 params.create_dataset(key, data=value) 

651 

652 sample = _nx_group(entry, "sample", "NXsample") 

653 sample.create_dataset("name", data=title or "calibration") 

654 

655 for name, x, y, units, y_name, description in spectra: 

656 x_name = _axis_name_for_units(units) 

657 _nx_data(entry, name, y_name, [x_name], {x_name: x, y_name: y}, 

658 units={x_name: units}, description=description) 

659 

660 doc = {"metadata": metadata or {}, **_laser_zero_info(calmodel), 

661 "model": calmodel.to_dict()} 

662 _nx_note(entry, "calibration_model", doc, h5) 

663 

664 _nx_data(entry, "calibration_curve", "calibrated_cm1", ["uncalibrated_cm1"], 

665 {"uncalibrated_cm1": grid, "calibrated_cm1": calibrated}, 

666 units={"uncalibrated_cm1": "cm-1", "calibrated_cm1": "cm-1"}, 

667 description="End-to-end x calibration: uncalibrated -> calibrated Raman shift.") 

668 

669 x_components = [c for c in getattr(calmodel, "components", []) 

670 if c.to_dict().get("type") != "YCalibrationComponent"] 

671 for idx, comp in enumerate(x_components): 

672 name = f"calibration_x_{idx}" if len(x_components) > 1 else "calibration_x" 

673 stage = stage_curves.get(id(comp)) 

674 _nx_calibration_group( 

675 inst, name, comp.to_dict(), h5, 

676 curve=(stage[0], stage[1]) if stage else None, 

677 curve_units=(stage[2], stage[3]) if stage else None) 

678 

679 if ycal_component is not None: 

680 ycal_group = _nx_calibration_group(inst, "calibration_y", ycal_component.to_dict(), h5) 

681 if y_error is not None: 

682 ycal_group.attrs["curves_error"] = y_error 

683 if y_data is not None: 

684 y_x, measured, certificate, factor = y_data 

685 axis_units = {"calibrated_cm1": "cm-1"} 

686 _nx_data(entry, "calibration_curve_y", "intensity_factor", ["calibrated_cm1"], 

687 {"calibrated_cm1": y_x, "intensity_factor": factor}, units=axis_units, 

688 description="y calibration factor applied by " 

689 "YCalibrationComponent.process: certificate response / " 

690 "measured response (0 where unusable).") 

691 _nx_data(entry, "calibration_curve_y_measured", "measured_response", 

692 ["calibrated_cm1"], 

693 {"calibrated_cm1": y_x, "measured_response": measured}, 

694 units=axis_units, 

695 description="Measured SRM response model (denominator of the factor).") 

696 if certificate is not None: 

697 _nx_data(entry, "calibration_curve_y_certificate", "certificate_response", 

698 ["calibrated_cm1"], 

699 {"calibrated_cm1": y_x, "certificate_response": certificate}, 

700 units=axis_units, 

701 description="Certificate response curve (numerator of the factor).") 

702 return filename
