1"""Relative-intensity calibration: fit the measured reference with the 

2certificate's own functional form. 

3 

4The measured SRM/LED reference is fitted with a polynomial of the certificate's 

5order (polynomial certificates) or the certificate's log-Gaussian (the SRM glass 

6certificates), seeded by the certificate's own parameters. The analytic fit 

7denoises the reference without smoothing. Driven with synthetic references of 

8the certificate's shape plus noise, so the outcome is deterministic and fast. 

9""" 

10 

11import numpy as np 

12import pytest 

13 

14from ramanchada2.protocols.calibration.interpolators import ( 

15 CustomPChipInterpolator, 

16 ParametricModel, 

17) 

18from ramanchada2.protocols.calibration.ycalibration import ( 

19 CertificatesDict, 

20 YCalibrationComponent, 

21) 

22from ramanchada2.spectrum import Spectrum 

23 

24POLY_CERT = ("785", "NIST785_SRM2241") # A0..A5 polynomial 

25LOG_GAUSS_CERT = ("532", "NIST532_SRM2242a") # H*exp(...)+m*x+b 

26 

27 

28def _noisy_reference(cert, scale=1000.0, noise=0.03, seed=0): 

29 """A measured reference of the certificate's shape, scaled and noised.""" 

30 lo, hi = cert.raman_shift 

31 x = np.linspace(lo, hi, 500) 

32 truth = np.asarray(cert.Y(x), dtype=float) 

33 rng = np.random.default_rng(seed) 

34 y = truth * scale + rng.normal(0, noise * np.max(truth) * scale, x.size) 

35 return x, y, truth

38def test_certificate_polynomial_order(): 

39 poly = CertificatesDict.load(*POLY_CERT) 

40 assert poly.polynomial_order == 5 

41 log_g = CertificatesDict.load(*LOG_GAUSS_CERT) 

42 assert log_g.polynomial_order is None 

43 # helpers parse the certificate's own parameters 

44 assert log_g.param_names == ["H", "w", "ro", "x0", "m", "b"] 

45 assert len(log_g.param_values) == 6

48@pytest.mark.parametrize("wl,key", [POLY_CERT, LOG_GAUSS_CERT]) 

49def test_reference_fitted_with_certificate_form(wl, key): 

50 cert = CertificatesDict.load(wl, key) 

51 x, y, truth = _noisy_reference(cert) 

52 ycal = YCalibrationComponent(int(wl), Spectrum(x=x, y=y), cert) 

53 

54 # the measured reference is now an analytic model of the certificate's form 

55 assert isinstance(ycal.model, ParametricModel) 

56 

57 fit = np.asarray(ycal.model(x), dtype=float) 

58 # the fit is far smoother than the raw measured reference (denoised)... 

59 raw_p2p = np.std(np.diff(y / np.max(np.abs(y)))) 

60 fit_p2p = np.std(np.diff(fit)) 

61 assert fit_p2p < raw_p2p / 5 

62 # ...and recovers the noise-free certificate shape (normalized) 

63 assert np.max(np.abs(fit - truth / np.max(truth))) < 0.1 

64 assert np.all(np.isfinite(fit))
